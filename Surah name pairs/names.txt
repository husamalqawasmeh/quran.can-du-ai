        const links = rawData.map(item => {
            const parts = item.pair.split('·').map(s => s.trim());
            return { left: parts[1], right: parts[0], count: item.count };
        });

        const leftNames = [...new Set(links.map(l => l.left))];
        const rightNames = [...new Set(links.map(l => l.right))];

        function render() {
            const leftCol = document.getElementById('left-column');
            const rightCol = document.getElementById('right-column');
            const svg = document.getElementById('connections-svg');
            const container = document.getElementById('canvas-container');

            leftCol.innerHTML = '<h3 class="text-center font-bold text-amber-400 border-b border-slate-800 pb-2 mb-2">الطرف الأيمن</h3>';
            rightCol.innerHTML = '<h3 class="text-center font-bold text-amber-400 border-b border-slate-800 pb-2 mb-2">الطرف الأيسر</h3>';
            svg.innerHTML = '';

            const leftElements = {};
            const rightElements = {};

            leftNames.forEach(name => {
                const div = document.createElement('div');
                div.className = 'bg-slate-900 border border-slate-700 p-2 rounded text-center text-xs font-medium shadow text-slate-200';
                div.textContent = name;
                leftCol.appendChild(div);
                leftElements[name] = div;
            });

            rightNames.forEach(name => {
                const div = document.createElement('div');
                div.className = 'bg-slate-900 border border-slate-700 p-2 rounded text-center text-xs font-medium shadow text-slate-200';
                div.textContent = name;
                rightCol.appendChild(div);
                rightElements[name] = div;
            });

            setTimeout(() => {
                const containerRect = container.getBoundingClientRect();
                svg.setAttribute('width', container.scrollWidth);
                svg.setAttribute('height', container.scrollHeight);

                links.forEach(link => {
                    const lEl = leftElements[link.left];
                    const rEl = rightElements[link.right];
                    if (!lEl || !rEl) return;

                    const lRect = lEl.getBoundingClientRect();
                    const rRect = rEl.getBoundingClientRect();

                    const x1 = rRect.left - containerRect.left + container.scrollLeft; 
                    const y1 = rRect.top + rRect.height / 2 - containerRect.top + container.scrollTop;
                    const x2 = lRect.left + lRect.width - containerRect.left + container.scrollLeft; 
                    const y2 = lRect.top + lRect.height / 2 - containerRect.top + container.scrollTop;

                    const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
                    const dx = Math.abs(x2 - x1) * 0.5;
                    const d = `M ${x1} ${y1} C ${x1 - dx} ${y1}, ${x2 + dx} ${y2}, ${x2} ${y2}`;

                    path.setAttribute('d', d);
                    path.setAttribute('fill', 'none');
                    path.setAttribute('stroke', '#38bdf8');
                    path.setAttribute('stroke-width', Math.max(1.5, Math.min(link.count * 0.4, 8)));
                    path.setAttribute('stroke-opacity', '0.6');
                    path.style.cursor = 'pointer';
                    path.style.transition = 'stroke-opacity 0.2s, stroke 0.2s';

                    path.addEventListener('mouseenter', (e) => {
                        path.setAttribute('stroke-opacity', '1');
                        path.setAttribute('stroke', '#f59e0b');
                        const tooltip = document.getElementById('tooltip');
                        tooltip.innerHTML = `الارتباط: <b>${link.right}</b> · <b>${link.left}</b><br>العدد: <b>${link.count}</b>`;
                        tooltip.style.display = 'block';
                        tooltip.style.left = (e.clientX + 10) + 'px';
                        tooltip.style.top = (e.clientY + 10) + 'px';
                    });

                    path.addEventListener('mousemove', (e) => {
                        const tooltip = document.getElementById('tooltip');
                        tooltip.style.left = (e.clientX + 10) + 'px';
                        tooltip.style.top = (e.clientY + 10) + 'px';
                    });

                    path.addEventListener('mouseleave', () => {
                        path.setAttribute('stroke-opacity', '0.6');
                        path.setAttribute('stroke', '#38bdf8');
                        document.getElementById('tooltip').style.display = 'none';
                    });

                    svg.appendChild(path);
                });
            }, 50);
        }

        window.addEventListener('load', render);
